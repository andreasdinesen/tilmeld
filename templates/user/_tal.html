{# Deltagere, gæster og afbud — samme opskrift på forsiden og på event-siden, så de to aldrig kan vise forskellige tal. `tal.count` er PLADSER (tilmeldte + deres gæster), fordi kapaciteten regner i mennesker i skoven. Opdelingen står nedenunder, så man kan se, hvad tallet består af — det var netop dét, man ikke kunne før. #} {% macro tal_linje(tal, ev, stor=false) %}
{{ tal.count }} / {{ ev.expected_count }} deltagere
{{ opdeling(tal, ev) }} {% endmacro %} {# Kun opdelingen — »8 tilmeldt · 6 gæster · 2 afbud«. Egen makro, fordi admin-siderne skriver hovedtallet med deres egne ord (»Deltagere: 14 / 22 forventet«), men skal regne på PRÆCIS det samme. Det var de holdt op med: admin trak antallet af tilmeldinger fra antallet af PLADSER, og på en jagt med gæster er den forskel negativ — så forsvandt afbuddene ud af oversigten. #} {% macro opdeling(tal, ev) %} {% set dele = [] %} {% if ev.allow_guests %} {% set _ = dele.append('%d tilmeldt' % tal.signups) %} {% set _ = dele.append('%d gæst%s' % (tal.guests, '' if tal.guests == 1 else 'er')) %} {% endif %} {% if tal.declined %} {% set _ = dele.append('%d afbud' % tal.declined) %} {% endif %} {% if tal.waitlist %} {% set _ = dele.append('%d på venteliste' % tal.waitlist) %} {% endif %} {% if dele %}
{{ dele|join(' · ') }}
{% endif %} {% endmacro %}
